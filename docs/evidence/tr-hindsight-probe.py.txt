"""Does bar_total_return leak hindsight? Read-only measurement on the dev stack. Not repo code.

    MSYS_NO_PATHCONV=1 docker compose run --rm -v "$(pwd -W)/docs/evidence:/e:ro" -e PYTHONPATH=/srv \
        worker python /e/tr-hindsight-probe.py.txt

1. bar_as_of(XLK, 2025-06-02, as_of=2025-07-01, include_backfill=True): c_adj against total_return.
2. A bar with dividends after it: is the stored Adj Close the bar's Close times the product, over every
   LATER ex-date E up to the fetch, of (1 - dividend_E / Close_{E-1})? (Yahoo's CRSP-style factor,
   ASSUMED as the hypothesis being tested.) And what bar_as_of returns at an as_of right after the bar.
3. Which basis yfinance's Dividends are on: at each XLK/XLE/XLY/XLU/XLB ex-date BEFORE the 2025-12-05 2:1
   split, the factor implied by the Adj Close series against 1 - div/Close_prev (split-adjusted Close)
   and against 1 - div/(2 x Close_prev) (as-traded Close).
4. Capital Gains: every nonzero value in the 2-year batch, and whether the Adj Close factor on that date
   includes it.
One 24-ticker daily batch (2y, actions on), through the real client. Writes nothing.
"""

import asyncio
from datetime import UTC, date, datetime

import pandas as pd

from app.ingest.bars import DAILY, NEW_YORK
from app.ingest.yfinance_client import YFinanceClient
from app.services import bars_read, bars_store
from app import settings
from sqlalchemy.ext.asyncio import create_async_engine


def et(*a):  # type: ignore[no-untyped-def]
    return datetime(*a, tzinfo=NEW_YORK)


async def main() -> None:
    engine = create_async_engine(settings.database_url())
    symbols = await bars_store.load_symbols(engine)
    tickers = [m.provider_symbol for m in symbols]
    now = datetime.now(UTC)
    batch = await YFinanceClient(max_batch=24).fetch(
        tickers, DAILY, today=now.astimezone(NEW_YORK).date(), period="2y"
    )
    f = batch.frame
    print(f"fetched {now.isoformat()}")

    def s(t: str, col: str) -> pd.Series:
        return f[(t, col)].dropna()

    # --- 1 -------------------------------------------------------------------------------------
    b = await bars_read.bar_as_of(engine, "XLK", date(2025, 6, 2), as_of=et(2025, 7, 1), include_backfill=True)
    print("\n== 1. bar_as_of(XLK, 2025-06-02, as_of=2025-07-01, include_backfill=True)")
    print(f"   c_raw {b.c_raw}  split_factor {b.split_factor}  c_adj {b.c_adj}  total_return {b.total_return}"
          f" (revision {b.total_return_revision}, available_at {b.available_at.astimezone(NEW_YORK)})")
    print(f"   c_adj / total_return = {float(b.c_adj / b.total_return):.4f}")
    close = float(s("XLK", "Close").loc[pd.Timestamp("2025-06-02")])
    adj = float(s("XLK", "Adj Close").loc[pd.Timestamp("2025-06-02")])
    print(f"   provider today: Close (split-adjusted) {close}, Adj Close {adj}; stored c_raw / Close = "
          f"{float(b.c_raw) / close:.4f}; Adj Close / Close = {adj / close:.6f}")

    # --- 2 -------------------------------------------------------------------------------------
    def check_dividends(t: str, d: date, as_of: datetime) -> None:
        cl, ad, dv = s(t, "Close"), s(t, "Adj Close"), f[(t, "Dividends")].fillna(0)
        later = [(ts, float(v)) for ts, v in dv.items() if v and ts.date() > d]
        predicted = 1.0
        for ts, v in later:
            prev = cl.loc[: ts - pd.Timedelta(days=1)].iloc[-1]
            predicted *= 1 - v / float(prev)
        actual = float(ad.loc[pd.Timestamp(d)]) / float(cl.loc[pd.Timestamp(d)])
        print(f"   {t} {d}: {len(later)} later ex-dates up to the fetch "
              f"({', '.join(f'{ts.date()} {v:g}' for ts, v in later)})")
        print(f"      Adj Close / Close today {actual:.6f}; product of (1 - div/Close_prev) over them {predicted:.6f}; "
              f"difference {abs(actual - predicted):.2e}")
        got = asyncio.get_event_loop()
        return later

    print("\n== 2. the stored total-return level contains dividends paid after the bar")
    later = check_dividends("SPY", date(2025, 3, 3), et(2025, 3, 6))
    b2 = await bars_read.bar_as_of(engine, "SPY", date(2025, 3, 3), as_of=et(2025, 3, 6, 12), include_backfill=True)
    print(f"   bar_as_of(SPY, 2025-03-03, as_of=2025-03-06 12:00 ET): c_raw {b2.c_raw}, total_return {b2.total_return}, "
          f"total_return / c_raw = {float(b2.total_return / b2.c_raw):.6f}")
    print("      (at that as_of, none of those later ex-dates had happened: the read hands a 2025 replay a 2026 level)")
    check_dividends("XLK", date(2025, 6, 2), et(2025, 7, 1))

    # --- 3 -------------------------------------------------------------------------------------
    print("\n== 3. the basis of yfinance's Dividends (ex-dates before the 2025-12-05 2:1 splits)")
    for t in ("XLK", "XLE", "XLY", "XLU", "XLB"):
        cl, ad, dv = s(t, "Close"), s(t, "Adj Close"), f[(t, "Dividends")].fillna(0)
        for ts, v in dv.items():
            if not v or ts.date() >= date(2025, 12, 5):
                continue
            i = cl.index.get_loc(ts)
            prev_ts = cl.index[i - 1]
            implied = (ad.loc[prev_ts] / cl.loc[prev_ts]) / (ad.loc[ts] / cl.loc[ts])
            split_basis = 1 - v / cl.loc[prev_ts]
            traded_basis = 1 - v / (2 * cl.loc[prev_ts])
            print(f"   {t} ex {ts.date()} div {v:g}: implied factor {implied:.6f}; 1 - div/Close_prev (split-adjusted) "
                  f"{split_basis:.6f}; 1 - div/(2 Close_prev) (as traded) {traded_basis:.6f}")

    # --- 4 -------------------------------------------------------------------------------------
    print("\n== 4. Capital Gains in the 2-year batch")
    found = 0
    for t in tickers:
        if (t, "Capital Gains") not in f.columns:
            print(f"   {t}: no Capital Gains column")
            continue
        cg = f[(t, "Capital Gains")].fillna(0)
        for ts, v in cg.items():
            if v:
                found += 1
                cl, ad = s(t, "Close"), s(t, "Adj Close")
                i = cl.index.get_loc(ts)
                prev_ts = cl.index[i - 1]
                implied = (ad.loc[prev_ts] / cl.loc[prev_ts]) / (ad.loc[ts] / cl.loc[ts])
                dv = float(f[(t, "Dividends")].fillna(0).loc[ts])
                print(f"   {t} {ts.date()} capital gain {v:g} (dividend same day {dv:g}): implied factor {implied:.6f}; "
                      f"1 - div/P {1 - dv / cl.loc[prev_ts]:.6f}; 1 - (div+cg)/P {1 - (dv + v) / cl.loc[prev_ts]:.6f}")
    print(f"   nonzero Capital Gains values: {found} across {len(tickers)} symbols")
    await engine.dispose()


asyncio.run(main())
